```python
import hashlib, json, subprocess, sys
from decimal import Decimal
from pathlib import Path

p = Path('evidence/diagnostic-cost-admissibility-20261015')
r = json.loads((p / 'receipt.json').read_text())
source = subprocess.check_output(['git', 'show', r['specimen']['commit'] + ':' + r['specimen']['path']])
assert hashlib.sha256(source).hexdigest() == r['specimen']['sha256']
tool = Path(r['quote_checker']['path']).read_bytes()
assert hashlib.sha256(tool).hexdigest() == r['quote_checker']['sha256']
rows = r['reported_resource_components']
assert sum(Decimal(x['wall_seconds']) for x in rows) == Decimal('78.4')
assert sum(Decimal(x['user_cpu_seconds']) for x in rows) == Decimal('61.0')
run = subprocess.run([sys.executable, r['quote_checker']['path'], str(p / 'quote-input.json')], capture_output=True, text=True)
assert run.returncode == 1
out = json.loads(run.stdout)
assert out == json.loads((p / 'quote-result.json').read_text())
assert out['status'] == 'INCOMPLETE_QUOTE' and out['arithmetic'] is None
for flag in ('offer_authorized', 'funding_established', 'settled_payment_established',
             'human_earnings_established', 'poverty_reduction_established', 'financial_authority'):
    assert out[flag] is False
print('PASS source-specific cost-evidence case; no cost or earnings inferred')
```
